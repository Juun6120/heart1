  <script>

    const namesByDate = {

      "2006.01.20": "饶俊",

      "2007.08.30": "童晨子",

      "2008.07.20": "傅结柔"

    };


    const form = document.querySelector("#date-form");

    const input = document.querySelector("#date-input");

    const heart = document.querySelector("#heart");

    const hint = document.querySelector("#hint");

    const result = document.querySelector("#result");

    const colors = ["#d66f82", "#c95e76", "#df8290", "#bd6379", "#e3919b"];


    function seededRandom(seed) {

      let value = seed >>> 0;

      return () => {

        value = (value * 1664525 + 1013904223) >>> 0;

        return value / 4294967296;

      };

    }


    function showHeart(date) {

      const name = namesByDate[date];

      if (!name) {

        heart.replaceChildren();

        heart.setAttribute("aria-label", "没有找到对应日期的名字");

        result.textContent = "";

        hint.textContent = "没有找到这个日期，请检查后再试试。";

        hint.classList.add("error");

        return;

      }


      hint.textContent = "日期格式：年.月.日";

      hint.classList.remove("error");

      result.textContent = `为「${name}」写下一颗心`;

      heart.setAttribute("aria-label", `由${name}组成的爱心`);


      const random = seededRandom([...date].reduce((sum, char) => sum + char.charCodeAt(0), 0));

      const fragment = document.createDocumentFragment();

      const columns = 22;

      const rows = 21;


      for (let row = 0; row < rows; row++) {

        for (let column = 0; column < columns; column++) {

          const x = (column / (columns - 1)) * 2.5 - 1.25;

          const y = 1.12 - (row / (rows - 1)) * 2.15;

          const shape = Math.pow(x * x + y * y - 1, 3) - x * x * y * y * y;

          if (shape > 0) continue;


          const word = document.createElement("span");

          word.className = "heart-word";

          word.textContent = name;

          word.style.left = `${(column / (columns - 1)) * 100}%`;

          word.style.top = `${(row / (rows - 1)) * 100}%`;

          word.style.setProperty("--angle", `${Math.round((random() - 0.5) * 52)}deg`);

          word.style.setProperty("--opacity", (0.72 + random() * 0.28).toFixed(2));

          word.style.setProperty("--color", colors[Math.floor(random() * colors.length)]);

          word.style.setProperty("--delay", `${(random() * 0.22).toFixed(2)}s`);

          fragment.append(word);

        }

      }


      heart.replaceChildren(fragment);

    }


    form.addEventListener("submit", (event) => {

      event.preventDefault();

      const date = input.value.trim().replace(/[／/．。]/g, ".").replace(/\s+/g, "");

      input.value = date;

      showHeart(date);

    });


    input.addEventListener("input", () => {

      if (hint.classList.contains("error")) {

        hint.textContent = "日期格式：年.月.日";

        hint.classList.remove("error");

      }

    });


    showHeart("2006.01.20");

  </script>
